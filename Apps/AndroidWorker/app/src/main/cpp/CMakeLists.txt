cmake_minimum_required(VERSION 3.22.1)
project(ShareComputeAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(sc-android SHARED worker-jni.cpp)
target_include_directories(sc-android PRIVATE "${SC_ROOT}/native/split")
foreach(lib sc-worker-core ggml ggml-rpc ggml-cpu ggml-base)
  add_library(${lib} STATIC IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${SC_ROOT}/build/android/lib/lib${lib}.a")
endforeach()
target_link_libraries(sc-android PRIVATE sc-worker-core ggml ggml-rpc ggml-cpu ggml-base log android dl)
