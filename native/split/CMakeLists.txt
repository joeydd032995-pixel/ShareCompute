cmake_minimum_required(VERSION 3.20)
project(ShareComputeSplit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned, patched llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_RPC ON CACHE BOOL "" FORCE)
foreach(flag GGML_RPC_RDMA GGML_METAL GGML_ACCELERATE GGML_OPENMP GGML_NATIVE GGML_AVX GGML_AVX2 GGML_AVX512 GGML_FMA GGML_F16C GGML_BMI2 LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/llama-revision.txt" SC_REVISION)
add_library(sc-worker-core STATIC worker.cpp)
target_link_libraries(sc-worker-core PUBLIC ggml ggml-rpc ggml-cpu)
target_compile_definitions(sc-worker-core PRIVATE SC_REVISION="${SC_REVISION}")
option(SC_IOS "Build only the native library for the Swift app" OFF)
if(NOT SC_IOS)
  add_executable(sc-rpc-worker worker-main.cpp)
  target_link_libraries(sc-rpc-worker PRIVATE sc-worker-core)
  add_executable(sc-split-probe probe.cpp)
  target_include_directories(sc-split-probe PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
  target_compile_definitions(sc-split-probe PRIVATE SC_REVISION="${SC_REVISION}")
  target_link_libraries(sc-split-probe PRIVATE llama ggml-rpc ggml-cpu)
endif()
