cmake_minimum_required(VERSION 3.20)
project(ShareComputeSplit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned, patched llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_RPC ON CACHE BOOL "" FORCE)
foreach(flag GGML_RPC_RDMA GGML_METAL GGML_ACCELERATE GGML_OPENMP GGML_NATIVE GGML_AVX GGML_AVX2 GGML_AVX512 GGML_FMA GGML_F16C GGML_BMI2 LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
# Opt-in CPU instruction sets (speed experiment E4). The defaults above stay the portable baseline, so a
# build that passes neither option is byte-for-byte what it was. The kit and the release never use them; CI builds
# the ARM one as separate experimental apps (jobs android-arm and ios-arm). Each changes accumulation order, so a
# build using one may need its own token reference (F52 report).
#   -DSC_X86_AVX2=ON          AVX2 + FMA + F16C + BMI2 (x86-64 only; SIGILL on a CPU without them)
#   -DSC_ARM_ARCH=armv8.2-a+dotprod   passed to ggml as GGML_CPU_ARM_ARCH (arm64 only)
option(SC_X86_AVX2 "Enable AVX2, FMA, F16C and BMI2 in the CPU backend" OFF)
set(SC_ARM_ARCH "" CACHE STRING "GGML_CPU_ARM_ARCH value, e.g. armv8.2-a+dotprod")
if(SC_X86_AVX2)
  foreach(flag GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_BMI2)
    set(${flag} ON CACHE BOOL "" FORCE)
  endforeach()
endif()
if(SC_ARM_ARCH)
  set(GGML_CPU_ARM_ARCH "${SC_ARM_ARCH}" CACHE STRING "" FORCE)
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/llama-revision.txt" SC_REVISION)
# The ShareCompute commit, passed by build_split_runtime.py. "unknown" when built by hand, which is
# recorded honestly rather than guessed (F44).
if(NOT SC_BUILD_COMMIT)
  set(SC_BUILD_COMMIT "unknown")
endif()
add_library(sc-worker-core STATIC worker.cpp)
target_link_libraries(sc-worker-core PUBLIC ggml ggml-rpc ggml-cpu)
target_compile_definitions(sc-worker-core PRIVATE SC_REVISION="${SC_REVISION}" SC_BUILD_COMMIT="${SC_BUILD_COMMIT}")
option(SC_IOS "Build only the native library for the Swift app" OFF)
if(NOT SC_IOS)
  add_executable(sc-rpc-worker worker-main.cpp)
  target_link_libraries(sc-rpc-worker PRIVATE sc-worker-core)
  add_executable(sc-split-probe probe.cpp)
  target_include_directories(sc-split-probe PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
  target_compile_definitions(sc-split-probe PRIVATE SC_REVISION="${SC_REVISION}" SC_BUILD_COMMIT="${SC_BUILD_COMMIT}")
  target_link_libraries(sc-split-probe PRIVATE llama ggml-rpc ggml-cpu)
endif()
